function renderFileList() { const ul = document.getElementById('fileList'); ul.innerHTML = ''; for (const name in S.files) { const f = S.files[name]; // このファイルがキャッシュにあり、かつ visible かどうか const cached = S.overlayCache.find(c => c.sourceFile === name); const isVisible = cached && cached.visible; const isActive = name === S.activeFileName; const li = document.createElement('li'); // 表示中のものはボーダーの色を変える li.className = 'file-item' + (isVisible ? ' active' : ''); li.style.borderLeft = isVisible ? '4px solid #16a34a' : '4px solid #d1d5db'; li.dataset.name = name; const info = document.createElement('div'); info.className = 'file-item-info'; info.innerHTML = '
' + (isVisible ? '👁️ ' : '') + escHtml(shortName(name)) + '
' + '
' + f.width + '×' + f.height + ' EPSG:' + (f.epsg || '??') + '
'; const btn = document.createElement('button'); btn.className = 'del-btn'; btn.textContent = '✕'; // ✕ボタンクリック時に親要素のクリック（表示切り替え）が走らないようにする btn.onclick = (e) => { e.stopPropagation(); const cached = S.overlayCache.find(c => c.sourceFile === name); if (cached) deleteCache(cached.id); delete S.files[name]; renderFileList(); refreshSelects(); }; li.appendChild(info); li.appendChild(btn); ul.appendChild(li); } }
